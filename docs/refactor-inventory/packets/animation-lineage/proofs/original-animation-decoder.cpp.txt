struct Stop {AnimationDecodeFailure failure;std::uint32_t address;};
[[noreturn]] void stop(AnimationDecodeFailure failure,std::uint32_t address){throw Stop{failure,address};}
class Image {
public:
    Image(const std::uint8_t* data,std::size_t size):data_(data),size_(size){
        if(!data||size<64||u16(data)!=0x5a4d)stop(AnimationDecodeFailure::invalid_pe,0);
        const auto pe=u32(data+60);const auto header=file(pe,24);if(u32(header)!=0x4550||u16(header+4)!=0x14c)stop(AnimationDecodeFailure::invalid_pe,0);
        const auto optional=file(std::size_t(pe)+24,u16(header+20));if(u16(header+20)<224||u16(optional)!=0x10b)stop(AnimationDecodeFailure::invalid_pe,0);
        base_=u32(optional+28);export_rva_=u32(optional+96);export_size_=u32(optional+100);const auto section=std::size_t(pe)+24+u16(header+20);const auto sections=file(section,std::size_t(u16(header+6))*40);
        for(std::size_t n=0;n<u16(header+6);++n){const auto p=sections+n*40;const auto bytes=u32(p+16),offset=u32(p+20);file(offset,bytes);regions_.push_back({u32(p+12),bytes,offset});}
    }
    const std::uint8_t* range(std::uint32_t address,std::size_t bytes,std::size_t* offset=nullptr)const {if(address<base_)stop(AnimationDecodeFailure::unavailable_range,address);const auto rva=address-base_;const std::uint8_t* found=nullptr;std::size_t selected=0;for(const auto& r:regions_)if(rva>=r.rva&&std::uint64_t(rva-r.rva)+bytes<=r.bytes){if(found)stop(AnimationDecodeFailure::unavailable_range,address);selected=std::size_t(r.offset)+rva-r.rva;found=file(selected,bytes);}if(!found)stop(AnimationDecodeFailure::unavailable_range,address);if(offset)*offset=selected;return found;}
    std::pair<std::size_t,std::size_t> backing(std::uint32_t address)const{std::size_t offset;range(address,1,&offset);const auto rva=address-base_;for(const auto& r:regions_)if(rva>=r.rva&&rva-r.rva<r.bytes)return {offset,r.bytes-(rva-r.rva)};stop(AnimationDecodeFailure::unavailable_range,address);}
    std::uint32_t address(std::uint32_t rva)const{if(rva>std::numeric_limits<std::uint32_t>::max()-base_)stop(AnimationDecodeFailure::invalid_pe,rva);return base_+rva;}
    std::uint32_t export_named(const char* wanted)const{if(!export_rva_)stop(AnimationDecodeFailure::missing_export,0);const auto e=range(address(export_rva_),40);const auto count=u32(e+24),functions=u32(e+20);const auto names=range(address(u32(e+32)),std::uint64_t(count)*4);const auto ordinals=range(address(u32(e+36)),std::uint64_t(count)*2);for(std::size_t n=0;n<count;++n){auto cursor=address(u32(names+n*4));std::size_t k=0;while(true){const char c=static_cast<char>(*range(cursor,1));if(c!=wanted[k])break;if(!c){const auto ordinal=u16(ordinals+n*2);if(ordinal>=functions)stop(AnimationDecodeFailure::invalid_pe,cursor);const auto slot=std::uint64_t(u32(e+28))+ordinal*4U;if(slot>0xffffffffU)stop(AnimationDecodeFailure::invalid_pe,cursor);const auto rva=u32(range(address(static_cast<std::uint32_t>(slot)),4));if(rva>=export_rva_&&std::uint64_t(rva)<std::uint64_t(export_rva_)+export_size_)stop(AnimationDecodeFailure::forwarded_export,address(rva));return address(rva);}if(cursor==0xffffffffU)stop(AnimationDecodeFailure::unavailable_range,cursor);++cursor;++k;}}stop(AnimationDecodeFailure::missing_export,address(export_rva_));}
    static std::uint16_t u16(const std::uint8_t* p){return std::uint16_t(p[0])|(std::uint16_t(p[1])<<8);}
    static std::uint32_t u32(const std::uint8_t* p){return std::uint32_t(p[0])|(std::uint32_t(p[1])<<8)|(std::uint32_t(p[2])<<16)|(std::uint32_t(p[3])<<24);}
    static float real(const std::uint8_t* p){const auto raw=u32(p);float f;std::memcpy(&f,&raw,4);return f;}
    static std::int32_t integer(const std::uint8_t* p){const auto raw=u32(p);std::int32_t i;std::memcpy(&i,&raw,4);return i;}
    static NinjaVector vector(const std::uint8_t* p){return {real(p),real(p+4),real(p+8)};}
private:
    const std::uint8_t* file(std::size_t at,std::size_t count)const{if(at>size_||count>size_-at)stop(AnimationDecodeFailure::invalid_pe,0);return data_+at;}
    struct Region {std::uint32_t rva,bytes,offset;};const std::uint8_t* data_;std::size_t size_;std::uint32_t base_=0,export_rva_=0,export_size_=0;std::vector<Region> regions_;
};
}
AnimationDecodeResult decode_sonic_animations(const std::uint8_t* bytes,std::size_t size,std::size_t first){try{
    const Image image(bytes,size);auto decoded=std::make_unique<DecodedPlayerAnimations>();decoded->next_resource_id=first;auto& d=*decoded;const auto key_bytes=std::make_shared<const std::vector<std::uint8_t>>(bytes,bytes+size);
    std::unordered_map<std::uint32_t,std::size_t> action_ids,object_ids,model_ids,node_ids,key_ids;std::map<std::uint32_t,std::size_t> motion_ids;std::unordered_map<std::uint32_t,std::size_t> motion_counts;std::unordered_map<std::uint32_t,unsigned> node_channels;std::vector<std::uint32_t> objects;
    const auto id=[&](auto& identities,std::uint32_t address)->std::optional<std::size_t>{if(!address)return {};auto found=identities.find(address);if(found!=identities.end())return found->second;if(d.next_resource_id==std::numeric_limits<std::size_t>::max())stop(AnimationDecodeFailure::resource_identity_overflow,address);const auto assigned=d.next_resource_id++;identities.emplace(address,assigned);return assigned;};
    const auto object_ref=[&](std::uint32_t address){const bool unseen=address&&object_ids.count(address)==0;auto assigned=id(object_ids,address);if(unseen)objects.push_back(address);return SceneResourceReference{assigned};};
    const auto locate=[&](std::size_t id,std::uint32_t address,std::size_t bytes,const char* kind){std::size_t offset;const auto p=image.range(address,bytes,&offset);d.locations[id]={address,offset,bytes,kind};return p;};
    d.export_source_address=image.export_named("___SONIC_ACTIONS");constexpr std::size_t export_slots=149;const auto exports=image.range(d.export_source_address,export_slots*4);d.sonic_actions.reserve(export_slots);
    for(std::size_t n=0;n<export_slots;++n){const auto pointer=Image::u32(exports+n*4);const auto assigned=id(action_ids,pointer);d.sonic_actions.push_back(AnimationResourceReference{assigned});if(assigned&&d.resources.actions.count(*assigned)==0){const auto p=locate(*assigned,pointer,8,"NJS_ACTION");d.resources.actions[*assigned]={object_ref(Image::u32(p)),AnimationResourceReference{id(motion_ids,Image::u32(p+4))}};}}
    for(const auto& recipe:sonic_animation_recipes)if(recipe.action_export){const auto ref=*d.sonic_actions.at(recipe.action_index);if(ref.id){const auto& action=d.resources.actions.at(*ref.id);if(action.motion->id){const auto motion_id=*action.motion->id;for(const auto& pair:motion_ids)if(pair.second==motion_id)motion_counts[pair.first]=std::max(motion_counts[pair.first],std::size_t(recipe.object_count));}}}
    for(const auto& pair:motion_ids){const auto address=pair.first;const auto assigned=pair.second;const auto p=locate(assigned,address,12,"NJS_MOTION");auto& m=d.resources.motions[assigned];m.frame_count=Image::u32(p+4);m.type=Image::u16(p+8);m.input_function=Image::u16(p+10);const auto node_address=Image::u32(p);const unsigned channels=((*m.type&4)|8)>>2;const auto node_id=id(node_ids,node_address);m.data=AnimationResourceReference{node_id};if(!node_id)continue;if(node_channels.count(node_address)&&node_channels.at(node_address)!=channels)stop(AnimationDecodeFailure::inconsistent_motion_layout,node_address);node_channels[node_address]=channels;const auto count=motion_counts[address];auto& array=d.resources.node_arrays[*node_id];if(array.size()>=count)continue;const auto nodes=locate(*node_id,node_address,count*channels*8,channels==3?"NJS_MDATA3":"NJS_MDATA2");for(std::size_t n=array.size();n<count;++n){PlayerAnimationNode node;const auto p=nodes+n*channels*8;if(channels==3)node.scale_pointer_bits_at_8=Image::u32(p+8);for(unsigned ch=0;ch<3;++ch){if(ch>=channels){node.keys[ch]=AnimationResourceReference{};node.counts[ch]=0;continue;}const auto address=Image::u32(p+ch*4),count=Image::u32(p+(channels+ch)*4);const auto key_id=id(key_ids,address);node.keys[ch]=AnimationResourceReference{key_id};node.counts[ch]=count;if(key_id){const auto backing=image.backing(address);const auto extent=std::max(1U,count);locate(*key_id,address,std::uint64_t(extent)*16,"NJS motion keys declared extent");d.resources.key_arrays[*key_id]={key_bytes,backing.first,backing.second};}}array.push_back(std::move(node));}}
    for(std::size_t n=0;n<objects.size();++n){const auto address=objects[n];const auto assigned=object_ids.at(address);const auto p=locate(assigned,address,52,"NJS_OBJECT");SceneGeometryObject o;o.evaluation_flags=Image::u32(p);o.model=SceneResourceReference{id(model_ids,Image::u32(p+4))};o.position=Image::vector(p+8);o.angles=std::array<std::int32_t,3>{Image::integer(p+20),Image::integer(p+24),Image::integer(p+28)};o.scale=Image::vector(p+32);o.child=object_ref(Image::u32(p+44));o.sibling=object_ref(Image::u32(p+48));d.objects.emplace(assigned,std::move(o));}
    d.resources.objects=&d.objects;return {AnimationDecodeFailure::none,0,std::move(decoded)};
}catch(const Stop& e){return {e.failure,e.address,{}};}}
}
